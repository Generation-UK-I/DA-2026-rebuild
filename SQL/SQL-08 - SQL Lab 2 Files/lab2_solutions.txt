1.sql: What is the total sales amount for each customer?

SELECT c.CustomerName, SUM(o.Total) AS TotalSales
FROM Orders o
JOIN Customers c ON o.CustomerID = c.CustomerID
GROUP BY o.CustomerID, c.CustomerName;

2.sql: Which product has the highest sales quantity?

SELECT p.ProductName,
SUM(o.Quantity) AS TotalQuantity
FROM Orders o Join Products p
ON p.ProductID = o.ProductID
GROUP BY o.ProductID
LIMIT 1;

3.sql What is the total revenue generated from each product?

SELECT p.ProductName, SUM(o.Total) AS TotalRevenue
FROM Orders o
JOIN Products p
ON o.ProductID = p.ProductID
GROUP BY p.ProductName;

4.sql How many orders were placed on each date?

SELECT OrderDate, count(OrderDate)
FROM Orders
GROUP BY OrderDate;

5.sql Who are the top 3 customers in terms of total sales amount?

SELECT c.CustomerName, SUM(o.Total) as TotalSales
FROM Customers c
JOIN Orders o ON o.CustomerID = c.CustomerID
GROUP BY o.CustomerID
ORDER BY TotalSales DESC
LIMIT 3;
